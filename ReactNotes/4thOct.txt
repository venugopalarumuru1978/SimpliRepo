    const Validate = () =>{

        const errors = {};
        let flag = true;
        if(empinfo.id===0)
        {
            errors.idErr = "Emp Number is Required!";
            flag = false;
        }
        if(empinfo.ename==="")
        {
            errors.enameErr = "Emp Name is Required!";
            flag = false;
        }
        if(empinfo.job==="")
        {
            errors.jobErr = "Emp Job is Required!";
            flag = false;
        }
        if(empinfo.salary===0.0)
        {
            errors.salErr = "Emp Sal is Required!";
            flag = false;
        }
        if(empinfo.email==="")
        {
            errors.emailErr = "Email is Required!";
            flag = false;
        }
        const regex = /^[\w-\.]+@([\w-]+\.)+[\w-]{2,4}$/i
        if(!regex.test(empinfo.email))
        {
            errors.emailInvalidErr = "Email is not Valid";
            flag = false;
        }

        if(empinfo.pswd==="")
        {
            errors.pwdErr = "Password is Required!";
            flag = false;
        }

        let pwdlength = empinfo.pswd.length;
        if(!(pwdlength>= 8 && pwdlength<=20))
        {
            errors.pwdlenErr = "Password must be b/w 8 - 20 chars only";
            flag = false;
        }
        console.log(errors);
        setFormErrors(errors);
        return flag;
    }
